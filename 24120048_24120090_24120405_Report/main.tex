% Dòng nhắc khi tệp chưa có: \chuaco{tên tệp}{người viết}
\newcommand{\chuaco}[2]{{\color{red}\textbf{[Chưa có nội dung]} Phần của #2, thêm tệp \texttt{sections/#1.tex}.}}

% \phan{tên tệp}{A|B|C}: chèn sections/<tên tệp>.tex nếu đã có.
\newcommand{\phan}[2]{%
	\IfFileExists{sections/#1.tex}%
		{\input{sections/#1.tex}}%
		{\par\noindent\chuaco{#1}{\csname nguoi#2\endcsname}\par}}

% \gop{itemize|dsso}{tên}: gộp các dòng \item trong sections/<tên>-A.tex,
% -B.tex, -C.tex thành một danh sách (dsso = danh sách đánh số).
\newlist{dsso}{enumerate}{1}
\setlist[dsso]{label=\textbf{\arabic*.}}
\newcommand{\gopmot}[2]{%
	\IfFileExists{sections/#1-#2.tex}%
		{\input{sections/#1-#2.tex}}%
		{\item \chuaco{#1-#2}{\csname nguoi#2\endcsname}}}
\newcommand{\gop}[2]{%
	\begin{#1}\gopmot{#2}{A}\gopmot{#2}{B}\gopmot{#2}{C}\end{#1}}
